BRUTE APPROACH :   TC = O(2n)+O(nlogn) , SC = O(1)


vector<int> findMissingRepeatingNumbers(vector<int> nums)
{
    int n = nums.size();
    int sum = 0;
    vector<int>ans;
    for(int i=0; i<n; i++)
    {
        sum = sum + nums[i];
    }
    int ogsum = n*(n+1)/2;
    
    int d = (ogsum - sum);
    
    sort(nums.begin(),nums.end());
    
    for(int i=0; i<n; i++)
    {
        if(nums[i]!=i+1)
        {
            if(d<0)
            {
                ans.push_back(i+1+(-1)*d);
                ans.push_back(i+1);
                return ans;
            }
            else
            {
                ans.push_back(i+1-d);
                ans.push_back(i+1);
                return ans;
            }
        }
    }
}

int main() 
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    vector<int>ans = findMissingRepeatingNumbers(nums);
    
    for(int i=0; i<2; i++)
    {
        cout<<ans[i]<<" ";
    }
}


OPTIMAL APPROACH :  TC = O(n) , SC = O(1)


vector<int> findMissingRepeatingNumbers(vector<int> nums)
{
    int n = nums.size();
    long long gsum = 0;
    long long sqgsum = 0;
    for(int i = 0; i<n; i++)
    {
        gsum = gsum + nums[i];
        sqgsum = (long long)sqgsum + ((long long)nums[i]*nums[i]);
    }
    long long ogsum = n*(n+1)/2;
    long long sqogsum = n*(n+1)*(2*n+1)/6;
    long long val1 = gsum - ogsum;
    long long val2 = (sqgsum - sqogsum)/val1;
    int repeating = (val1+val2)/2;
    int missing = (val2-val1)/2;
    
    return {repeating,missing};
}
