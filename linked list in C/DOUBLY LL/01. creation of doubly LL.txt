#include <stdio.h>
#include<stdlib.h>

struct node
{
    int data;
    struct node* next;
    struct node* prev;
};

struct node* head;
struct node* newnode;
struct node* temp;
struct node* prev;

struct node* create_doubly_LL()
{
    int choice = 1;
    head = NULL;
    prev = NULL;
    
    while(choice!=0)
    {
        newnode = (struct node*)malloc(sizeof(struct node));
        newnode->next = NULL;
        printf("enter data : ");
        scanf("%d",&newnode->data);
        
        if(head==NULL)
        {
            head = newnode;
            newnode->prev = NULL;
        }
        else
        {
            newnode->prev = temp;
            temp->next = newnode;
        }
        
        temp = newnode;
        printf("enter 1 to continue else 0 : ");
        scanf("%d",&choice);
    }

    return head;
}

void display(struct node* list)
{
    temp = list;
    while(temp!=NULL)
    {
        printf("%d",temp->data);
        printf(" ");
        temp = temp->next;
    }
    return;
}
int main() 
{
    create_doubly_LL();
    display(head);
    
}
