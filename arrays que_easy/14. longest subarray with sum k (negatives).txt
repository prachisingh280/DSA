#include <bits/stdc++.h>
using namespace std;

int longestSubarray(vector<int> &nums, int k)
{
    int n = nums.size();
    unordered_map<int,int>hash_map;
    int sum = 0;
    int max_len = 0;
    hash_map[0] = -1;
    
    for(int i=0; i<n; i++)
    {
        sum = sum + nums[i];
        if(sum == k)
        {
            max_len = max(max_len,i+1);
        }
        if(hash_map.find(sum-k) == hash_map.end())
        {
            hash_map[sum] = i;
        }
        else
        {
            max_len = max(max_len,i - hash_map[sum-k]);
        }
    }
    return max_len;
}
int main() 
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    int k;
    cin>>k;
    cout<<longestSubarray(nums,k);
}
