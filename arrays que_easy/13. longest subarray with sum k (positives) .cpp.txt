BRUTE APPROACH :       TC = O(n^2), SC = O(n)


int single_number(vector<int>& nums,int k)
{
    int n = nums.size();
    int len = 0; 
    
    for(int i=0; i<n; i++)
    {
        int summ = 0;
        for(int j=i; j<n; j++)
        {
            summ = summ + nums[j];
            if(summ == k)
            {
                len = max(len,j-i+1);
            }
        }
    }
    return len;
}
int main()
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    int k;
    cin>>k;
    
    cout<<single_number(nums,k);
}

BETTER APPROACH :    TC = O(n) , SC = O(n)

int longestSubarray(vector<int> &nums, int k)
{
    int n = nums.size();
    unordered_map<int,int>hash_map;
    int sum = 0;
    int max_len = 0;
    hash_map[0] = -1;
    
    for(int i=0; i<n; i++)
    {
        sum = sum + nums[i];
        if(sum == k)
        {
            max_len = max(max_len,i+1);
        }
        if(hash_map.find(sum-k) == hash_map.end())
        {
            hash_map[sum] = i;
        }
        else
        {
            max_len = max(max_len,i - hash_map[sum-k]);
        }
    }
    return max_len;
}

OPTIMAL APPROACH : TC = O(2n) , SC = O(1)

int longestSubarray(vector<int> &nums, int k)
{
    int n = nums.size();
    int j = 0;
    int sum = 0;
    int max_len = 0;
    
    for(int i=0; i<n; i++)
    {
        sum += nums[i];
        while(sum>k)
        {
            sum = sum - nums[j];
            j++;
        }
        if(sum==k)
        {
            max_len = max(max_len,i-j+1);
        }
        
    }
    return max_len;
}
