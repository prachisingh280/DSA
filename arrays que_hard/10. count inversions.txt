BRUTE APPROACH :   TC = O(n^2) , SC = O(1)


long long int numberOfInversions(vector<int> nums)
{
    int n = nums.size();
    int cnt = 0;
    
    for(int i=0; i<n; i++)
    {
        for(int j=i; j<n; j++)
        {
            if(nums[i]>nums[j])
            {
                cnt++;
            }
        }
    }
    
    return cnt;
}

int main() 
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    cout<<numberOfInversions(nums);
}
