ceil is nothing but lower bound. not more than that. same exact code!

floor is the value just less than or equal to the number we are searching. 
this is binary search, hence the TC = O(logn)

int floor(vector<int>&nums, int x)
{
    int n = nums.size();
    int l = 0;
    int h = n-1;
    int ans = n;
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(nums[mid]>x)
        { 
            h = mid-1;
        }
        else if(nums[mid]<x)
        {
            ans = mid;
            l = mid+1;
        }
        else
        {
            return nums[mid];
        }
    }
    if(ans == n)
    {
        return -1;
    }
    return nums[ans];
}
