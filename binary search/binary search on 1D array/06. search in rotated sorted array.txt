long long search(vector<long long>& nums, long long target)
{
    long long n = nums.size();
    long long l = 0;
    long long h = n-1;
    
    while(l<=h)
    {
        long long mid = (l+h)/2;
        if(nums[mid]==target)
        {
            return mid;
        }
        else if(nums[l]<=nums[mid])
        {
            if(nums[l]<=target && target<=nums[mid])
            {
                h = mid-1;
            }
            else
            {
                l = mid+1;
            }
        }
        else if(nums[mid]<=nums[h])
        {
            if(nums[mid]<=target && target<=nums[h])
            {
                l = mid+1;
            }
            else
            {
                h = mid-1;
            }
        }
    }
    
    return -1;
}

int main() 
{
    long long n;
    cin>>n;
    vector<long long>nums(n);
    
    for(long long i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    long long target;
    cin>>target;
    
    cout<<search(nums,target);
}
