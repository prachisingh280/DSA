BRUTE APPROACH : TC = O(r*c),   SC = O(1)

int rowWithMax1s(vector<vector<int>>& mat)
{
    int r = mat.size();
    int c = mat[0].size();
    int index = -1;
    int max_cnt = -1;
    
    for(int i=0; i<r; i++)
    {
        int cnt = 0;
        for(int j=0; j<c; j++)
        {
            if(mat[i][j]==1)
            {
                cnt++;
            }
        }
        
        if(cnt>max_cnt)
        {
            index = i;
            max_cnt = cnt;
        }
    }
    
    return index;
}
int main() 
{
    int r;
    int c;
    cin>>r>>c;
    vector<vector<int>>mat(r,vector<int>(c));
    
    for(int i=0; i<r; i++)
    {
        for(int j=0; j<c; j++)
        {
            cin>>mat[i][j];
        }
    }
    
    cout<<rowWithMax1s(mat);
}

int matrix_traversals(vector<int>&mat, int c)
{
    int r = mat.size();
    int index = lower_bound(mat.begin(), mat.end(),1) - mat.begin();
    return c - index;
}
int rowWithMax1s(vector<vector<int>>& mat)
{
    int r = mat.size();
    int c = mat[0].size();
    int index = -1;
    int max_ones = 0;
    
    for(int i=0; i<r; i++)
    {
        int ones = matrix_traversals(mat[i], c);
        if(ones>max_ones)
        {
            index = i;
            max_ones = ones;
        }
    }
    
    return index;
}
