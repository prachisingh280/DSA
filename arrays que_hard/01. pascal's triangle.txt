ALL THREE FORMS:
1. GIVEN ROW AND COLUMN, TELL THE NUMBER OCCURING AT THOSE CORDINATES.


TC = O(column) , SC = O(1)

void generate(int row,int column)
{
    int res = 1;
    for(int i=0; i<column; i++)
    {
        res = res*(row-i);
        res = res/(i+1);
    }
    
    cout<<res<<"\n";
}

int main() 
{
    
    int row;
    int column;
    
    cin>>row>>column;
    generate(row-1, column-1);
}
