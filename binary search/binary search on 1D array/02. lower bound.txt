FIRST APPROACH USES ITERATIVE SEARCHING WHICH USES O(n) TIME COMLEXITY.
THE OPTIMAL APPROACH USES BINARY SEARCH WHICH HAS A TC = O(log n).
WE CAN ALSO USE AN STL TO FIND LOWER BOUND ie. lower_bound(arr.begin(),arr.end(),n);

# BINARY SEARCH
int lowerbound(vector<int>&arr, int x)
{
    int n = arr.size();
    int l = 0;
    int h = n-1;
    int ans = n;
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(x>arr[mid])
        {
            l = mid+1;
        }
        else 
        {
            ans = mid;
            h = mid-1;
        }
        
    }
    return ans;
}
int main()
{
    int n;
    cin>>n;
    vector<int>arr(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>arr[i];
    }
    
    int x;
    cin>>x;
    
    cout<<lowerbound(arr,x);
}
