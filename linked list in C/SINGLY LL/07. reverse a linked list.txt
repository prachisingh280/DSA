void reverse()
{
    struct node* nextnode;
    struct node* curr = head;
    struct node* prev = NULL;
    
    while(nextnode!=NULL)
    {
        nextnode = curr->next;
        curr->next = prev;
        prev = curr;
        curr = nextnode;
    }

    head = prev;
}
