#include<stdio.h>
#include<stdlib.h>

struct node
{
    int coeff;
    int exp;
    struct node* next;
};

struct node* head = NULL;
struct node* newnode;
struct node* temp;

void creation()
{
    int n;

    printf("enter the number of terms : ");
    scanf("%d",&n);

    for(int i=0; i<n; i++)
    {
        newnode = (struct node*)malloc(sizeof(struct node));
        printf("enter coefficient : ");
        scanf("%d",&newnode->coeff);
        printf("enter exponent : ");
        scanf("%d",&newnode->exp);
        newnode->next = NULL;

        if(head == NULL)
        {
            head = newnode;
            temp = newnode;
        }
        else
        {
            temp->next = newnode;
            temp = newnode;
        }
    }
}

void display()
{
    temp = head;

    while(temp!=NULL)
    {
        printf("%dx^%d",temp->coeff, temp->exp);
        if(temp->next!=NULL)
        {
            printf(" + ");
        }

        temp = temp->next;
    }
}

int main()
{
    creation();
    display();
}
