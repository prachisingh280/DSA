

int findMin(vector<int>& nums)
{
    int n = nums.size();
    int l = 0;
    int h = n-1;
    int ans = 0;
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(nums[l]<=nums[mid])
        {
            if(nums[ans] >= nums[mid])
            {
                ans = l;
            }
            l = mid+1;
        }
        
        else if(nums[mid]<=nums[h])
        {
            if(nums[ans]>=nums[h])
            {
                ans = mid;
            }
            h = mid-1;
        }
    }
    return ans;
}

int main() 
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    cout<<findMin(nums);
}
