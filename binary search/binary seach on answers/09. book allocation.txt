#include <bits/stdc++.h>
using namespace std;

int possible(vector<int>&nums, int pages)
{
    int n = nums.size();
    int cnt = 1;
    int sum = 0;
    
    for(int i=0; i<n; i++)
    {
        if((sum + nums[i])<=pages)
        {
            sum = sum + nums[i];
        }
        else
        {
            cnt++;
            sum = nums[i];
        }
    }
    
    return cnt;
}

int findPages(vector<int> &nums, int m)
{
    int n = nums.size();
    if(n<m)
    {
        return -1;
    }
    int ans = -1;
    int l = *max_element(nums.begin(),nums.end());
    int sum = 0;
    for(int i=0; i<n; i++)
    {
        sum = sum + nums[i];
    }
    
    int h = sum; 
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(possible(nums,mid)>m)
        {
            ans = mid;
            l = mid+1;
        }
        else
        {
            h = mid-1;
        }
    }
    return ans+1;
}

int main() 
{
    int n;
    cin>>n;
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    int m;
    cin>>m;
    
    cout<<findPages(nums,m);
}
