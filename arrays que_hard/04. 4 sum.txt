BRUTE APPROACH :    TC = O(n^3 + log m){m is the no. of elements in the set} , SC = O(n) + O(vectors*2) {2 times because one for set and other for returning the ans}

#include <bits/stdc++.h>
using namespace std;

vector<vector<long long>> fourSum(vector<long long>& nums, long long target)
{
    long long n = nums.size();
    set<vector<long long>>vt;
    for(long long i=0; i<n; i++)
    {   
        for(long long j = i+1; j<n; j++)
        { 
            unordered_set<long long>st;
            for(long long k=j+1; k<n; k++)
            {
                long long sum = nums[i]+nums[j];
                sum += nums[k];
                long long l = target - (sum);
                if(st.find(l)!=st.end())
                {
                    vector<long long>temp = {nums[i],nums[j],nums[k],(long long)(l)};
                    sort(temp.begin(),temp.end());
                    vt.insert(temp);
                }
                st.insert(nums[k]);
            }
            
        }
    }
    
    vector<vector<long long>>ans(vt.begin(),vt.end());
    return ans;
}

int main() 
{
    long long n;
    cin>>n;
    vector<long long>nums(n);
    
    for(long long i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    long long target;
    cin>>target;
    
    vector<vector<long long>>ans = fourSum(nums, target);
    
    for(auto it : ans)
    {
        for(auto ele : it)
        {
            cout<<ele<<" ";
        }
        cout<<", ";
    }
}


OPTIMAL APPROACH :  TC = O(n^3) + O(logn){sorting the nums}, SC = O(1){temp takes a space of 4 vectors which equally good as constant}


vector<vector<int>> fourSum(vector<int>& nums, int target)
{
    int n = nums.size();
    vector<vector<int>>ans;
    sort(nums.begin(),nums.end());
    
    for(int i=0; i<n; i++)
    {
        if(i>0 && nums[i] == nums[i-1]) continue;
        
        for(int j=i+1; j<n; j++)
        {
            if(j>i+1 && nums[j] == nums[j-1]) continue;
            int k = j+1;
            int l = n-1;
            while(k<l)
            {
                long long sum = nums[i] + nums[j];
                sum += nums[k];
                sum += nums[l];
                
                if(sum<target)
                {
                    k++;
                }
                else if(sum>target)
                {
                    l--;
                }
                else
                {
                    vector<int>temp = {nums[i], nums[j], nums[k], nums[l]};
                    ans.push_back(temp);
                    k++;
                    l--;
                    while(k<l && nums[k]==nums[k-1]) {k++;}
                    while(k<l && nums[l]==nums[l+1]) {l--;}
                }
            }
        }
    }
    
    return ans;
}
