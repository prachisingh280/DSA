#include<stdio.h>
#include<stdlib.h>

struct node
{
    int data;
    struct node* next;
};

struct node* head = NULL;
struct node* newnode;
struct node* temp;

void creation()
{
    temp = head;
    int choice = 1;
    
    while(choice!=0)
    {
        newnode = (struct node*)malloc(sizeof(struct node));
        printf("enter data: ");
        scanf("%d",&newnode->data);
        newnode->next = head;
        
        if(head == NULL)
        {
            head = newnode;
        }
        else
        {
            temp->next = newnode;
        }
        temp = newnode;
        
        printf("1 to continue otherwise 0 : ");
        scanf("%d",&choice);   
    }
}

void display()
{
    temp = head;
    
    if(temp==NULL)
    {
        printf("list is empty");
        return;
    }
    
    do
    {
        printf("%d ",temp->data);
        temp = temp->next;
    }
    while(temp!=head);
    
}

int main()
{
    creation();
    display();
}
