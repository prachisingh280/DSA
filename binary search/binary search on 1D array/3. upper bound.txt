

int lowerbound(vector<int>&arr, int x)
{
    int n = arr.size();
    int l = 0;
    int h = n-1;
    int ans = n;
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(arr[mid]<=x)
        {
            l = mid+1;
        }
        else if(arr[mid]>x)
        {
            ans = mid;
            h = mid-1;
        }
    }
    return ans;
    
}
int main()
{
    int n;
    cin>>n;
    vector<int>arr(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>arr[i];
    }
    
    int x;
    cin>>x;
    
    cout<<lowerbound(arr,x);
}
