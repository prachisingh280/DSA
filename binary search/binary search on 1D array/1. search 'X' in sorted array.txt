RECURSIVE APPROACH :


int bs(vector<int>&nums, int low,int high, int target)
{
    if(low>high)
    return -1;
        
    int mid = (low+high)/2;
    
    if(target > nums[mid])
    {
       return bs(nums,mid+1,high,target);
    }
    else if(target < nums[mid])
    {
       return bs(nums,low,mid-1,target);
    }
    else 
    return mid;
}

int search(vector<int>& nums, int target)
{
    int n = nums.size();
    return bs(nums,0,n-1,target);
}

int main() 
{
    int n;
    cin>>n;
    
    vector<int>nums(n);
    
    for(int i=0; i<n; i++)
    {
        cin>>nums[i];
    }
    
    int target;
    cin>>target;
    
    cout<<search(nums,target);
} 


SIMPLE APPROACH :

int search(vector<int>& nums, int target)
{
    int n = nums.size();
    
    int l = 0;
    int h = n-1;
    
    while(l<=h)
    {
        int mid = (l+h)/2;
        if(target>nums[mid])
        {
            l = mid+1;
        }
        else if(target<nums[mid])
        {
            h = mid-1;
        }
        else
        {
            return mid;
        }
    }
    return -1;
}
